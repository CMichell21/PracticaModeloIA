print("========================================")
print("SE ESTA EJECUTANDO cantidad_caracteres.py")
print("========================================")
import pandas as pd
from sentence_transformers import SentenceTransformer


# ============================================================
# CONFIGURACIÓN
# ============================================================

ARCHIVO_ENTRADA = "textos_entrada.xlsx"
ARCHIVO_SALIDA = "resultado_tokens.xlsx"

COLUMNA_ID = "ID"
COLUMNA_TEXTO = "TEXTO"


# ============================================================
# CARGAR MODELO
# ============================================================

print("Cargando modelo...")

model = SentenceTransformer(
    "sentence-transformers/all-MiniLM-L6-v2"
)

# Obtener el máximo configurado por el modelo
MAX_TOKENS_MODELO = model.max_seq_length

print(f"Modelo cargado correctamente.")
print(f"Máximo de tokens del modelo: {MAX_TOKENS_MODELO}")


# ============================================================
# TOKENIZER DEL MODELO
# ============================================================

tokenizer = model.tokenizer


# ============================================================
# LEER EXCEL
# ============================================================

print()
print(f"Leyendo archivo: {ARCHIVO_ENTRADA}")

df = pd.read_excel(ARCHIVO_ENTRADA)

# Validar columnas
if COLUMNA_ID not in df.columns:
    raise ValueError(
        f"No existe la columna '{COLUMNA_ID}'"
    )

if COLUMNA_TEXTO not in df.columns:
    raise ValueError(
        f"No existe la columna '{COLUMNA_TEXTO}'"
    )


# ============================================================
# LISTA PARA GUARDAR RESULTADOS
# ============================================================

resultados = []


# ============================================================
# PROCESAR REGISTRO POR REGISTRO
# ============================================================

for _, fila in df.iterrows():

    id_texto = fila[COLUMNA_ID]
    texto = fila[COLUMNA_TEXTO]

    # --------------------------------------------------------
    # Validar texto
    # --------------------------------------------------------

    if pd.isna(texto):
        texto = ""

    texto = str(texto)

    # --------------------------------------------------------
    # Cantidad de caracteres
    # --------------------------------------------------------

    cantidad_caracteres = len(texto)

    # --------------------------------------------------------
    # Obtener TODOS los tokens
    # --------------------------------------------------------

    tokens = tokenizer.encode(
        texto,
        add_special_tokens=True,
        truncation=False
    )

    cantidad_tokens = len(tokens)

    tokens_truncados = tokenizer.encode(
    texto,
    add_special_tokens=True,
    truncation=True,
    max_length=MAX_TOKENS_MODELO
    )

    cantidad_tokens_truncado = len(tokens_truncados)

    texto_truncado = tokenizer.decode(
        tokens_truncados,
        skip_special_tokens=True
    )

    cantidad_caracteres_truncado = len(texto_truncado)
    # --------------------------------------------------------
    # Determinar si supera el máximo del modelo
    # --------------------------------------------------------

    supera_maximo = (
        cantidad_tokens > MAX_TOKENS_MODELO
    )

    # --------------------------------------------------------
    # Guardar resultado
    # --------------------------------------------------------

    resultados.append({
        "ID": id_texto,
        "CARACTERES": cantidad_caracteres,
        "TOKENS": cantidad_tokens,
        "CARACTERES_TRUNCADO": cantidad_caracteres_truncado,
        "TOKENS_TRUNCADO":cantidad_tokens_truncado,
        "SUPERA_MAXIMO": supera_maximo
    })

    # --------------------------------------------------------
    # Mostrar progreso
    # --------------------------------------------------------

    print(
        f"ID: {id_texto} | "
        f"Caracteres: {cantidad_caracteres} | "
        f"Tokens: {cantidad_tokens}"
    )


# ============================================================
# CREAR DATAFRAME DE RESULTADOS
# ============================================================

df_resultado = pd.DataFrame(resultados)


# ============================================================
# GUARDAR EXCEL
# ============================================================

df_resultado.to_excel(
    ARCHIVO_SALIDA,
    index=False
)


# ============================================================
# INFORMACIÓN FINAL
# ============================================================

print()
print("=" * 60)
print("PROCESAMIENTO TERMINADO")
print("=" * 60)

print(f"Registros procesados: {len(df_resultado)}")

print(f"Archivo generado: {ARCHIVO_SALIDA}")

print("=" * 60)
