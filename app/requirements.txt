import unittest
from unittest.mock import AsyncMock, patch

from fastapi import Request

from app.api.comparar import comparar


class TestCompararEndpoint(unittest.IsolatedAsyncioTestCase):

    @patch(
        "app.api.comparar.orquestador.ejecutar_proceso",
        new_callable=AsyncMock
    )
    async def test_comparar(self, mock_ejecutar):

        resultado_esperado = {
            "estado": "OK",
            "mensaje": "Proceso ejecutado correctamente"
        }

        mock_ejecutar.return_value = resultado_esperado

        request = Request({
            "type": "http",
            "method": "POST",
            "path": "/comparar",
            "headers": [],
            "query_string": b"",
            "server": ("testserver", 80),
            "client": ("testclient", 50000),
            "scheme": "http"
        })

        resultado = await comparar(request)

        self.assertEqual(
            resultado,
            resultado_esperado
        )

        mock_ejecutar.assert_awaited_once_with(request)


if __name__ == "__main__":
    unittest.main()
